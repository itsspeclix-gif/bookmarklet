/* ##########################################################################
 * §1  CORE BACKEND · BOOKMARKLET FORMS AUTH CLIENT
 * ##########################################################################
 *  How sign-in works (unchanged from v8.1, compatible with the v6 backend):
 *
 *    browser ──POST──▶ Google Form  (request id + JSON payload)
 *                         │
 *                    Apps Script  (checks Accounts sheet, writes reply)
 *                         │
 *    browser ◀──read── Response sheet (gviz JSONP query + CSV export, raced)
 *
 *  Every reply is HMAC-signed with a one-time secret that only this request
 *  knew, so a reply cannot be forged, replayed, or mixed up with another one.
 *
 *  Exposes: window.BookmarkletFormsAuth  (used by the app layer below)
 *  Nothing in this section touches the page UI.
 * ########################################################################## */
(function (global) {
    'use strict';
    const BUILD = '8.2';

    /* ---- 1a. Small helpers: base64url, random ids, CSV parser ------------- */
    function b64(bytes) {
        let s = '';
        for (const b of bytes)
            s += String.fromCharCode(b);
        return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }
    function unb64(s) {
        const t = s.replace(/-/g, '+').replace(/_/g, '/');
        return Uint8Array.from(atob(t + '='.repeat((4 - t.length % 4) % 4)), c => c.charCodeAt(0));
    }
    function randomKey() {
        return b64(global.crypto.getRandomValues(new Uint8Array(32)));
    }
    function randomId() {
        return Array.from(global.crypto.getRandomValues(new Uint8Array(16)), b => b.toString(16).padStart(2, '0')).join('');
    }
    function number(x, d, min, max) {
        x = Number(x);
        return Number.isFinite(x) ? Math.max(min, Math.min(max, x)) : d;
    }
    function clock() {
        return global.performance && typeof global.performance.now === 'function' ? global.performance.now() : Date.now();
    }
    function parseCSV(text) {
        text = text.replace(/^\uFEFF/, '');
        const rows = [];
        let row = [], cell = '', quoted = false;
        for (let i = 0; i < text.length; i++) {
            const c = text[i];
            if (c === '"') {
                if (quoted && text[i + 1] === '"') {
                    cell += '"';
                    i++;
                }
                else
                    quoted = !quoted;
            }
            else if (c === ',' && !quoted) {
                row.push(cell);
                cell = '';
            }
            else if ((c === '\r' || c === '\n') && !quoted) {
                if (c === '\r' && text[i + 1] === '\n')
                    i++;
                row.push(cell);
                rows.push(row);
                row = [];
                cell = '';
            }
            else
                cell += c;
        }
        if (quoted)
            throw new Error('Invalid CSV response.');
        if (cell || row.length) {
            row.push(cell);
            rows.push(row);
        }
        return rows;
    }
    /* ---- 1b. Error type + human-readable messages for every server code --- */
    class AuthError extends Error {
        constructor(code, message) {
            super(message || code);
            this.name = 'AuthError';
            this.code = code;
        }
    }
    const messages = {
        INVALID_CREDENTIALS: 'Incorrect username or password.',
        PASSWORD_NOT_SET: 'This account has no password in Accounts column E.',
        DUPLICATE_USERNAME: 'This username occurs more than once in Accounts. Remove the duplicate row.',
        ACCOUNT_BANNED: 'This account is banned.',
        ACCOUNT_DISABLED: 'This account is disabled.',
        INVALID_PROTECTION_SETTING: 'The protection column has an unsupported value.',
        INVALID_ORIGIN: 'Run this bookmark on a normal HTTPS website, not the downloaded builder, a new tab, or about:blank.',
        BROWSER_LOCKED: 'This account is locked to another browser/site. Ask the owner to reset its browser lock.',
        SESSION_REVOKED: 'Your session was revoked. Sign in again.',
        INVALID_SESSION: 'Your saved session is missing or expired.',
        RATE_LIMITED: 'Too many failed attempts. Try again after the configured login window.',
        STALE_REQUEST: 'The request was delayed or the device clock is incorrect. Try again.',
        SERVER_ERROR: 'The backend failed. Check Apps Script Executions.',
        TIMEOUT: 'The check timed out. No verified reply was accepted.',
        TRANSPORT_ERROR: 'The reply sheet could not be read from this page. A page policy, browser/network restriction, or sharing setting may be blocking it.',
        INVALID_REPLY: 'The reply is malformed, stale, for another request, or has an invalid signature.',
        CANCELLED: 'Check cancelled. A submission already received by Google may still be processed.'
    };
    function failure(code, detail) {
        return new AuthError(code, detail || messages[code]);
    }
    /* ---- 1c. The client --------------------------------------------------- */
    class BookmarkletFormsAuth {
        /* Validates the page (HTTPS only) and the backend config. */
        constructor(config) {
            if (!global.crypto || !global.crypto.subtle)
                throw failure('HTTPS_REQUIRED', 'Use a normal HTTPS website.');
            const loc = global.location;
            if (!loc || (loc.protocol && loc.protocol !== 'https:') ||
                typeof loc.origin !== 'string' || !/^https:\/\/[a-z0-9.-]+(?::[0-9]{1,5})?$/i.test(loc.origin) || loc.origin.length > 250)
                throw failure('INVALID_ORIGIN');
            if (!config || config.version !== 6 || !/^https:\/\/docs\.google\.com\/forms\/d\/(?:e\/)?[A-Za-z0-9_-]+\/formResponse$/.test(config.formPostUrl))
                throw failure('BAD_CONFIG', 'Use the configuration from Setup!B5.');
            if (!config.fields || !/^entry\.\d+$/.test(config.fields.requestId) || !/^entry\.\d+$/.test(config.fields.payload) ||
                !/^[A-Za-z0-9_-]+$/.test(config.responsesId) || !Number.isInteger(config.responsesGid) || config.responsesGid < 0)
                throw failure('BAD_CONFIG', 'Invalid Form field or response-sheet configuration.');
            this.config = Object.assign({}, config, {
                timeoutMs: number(config.timeoutMs, 45000, 10000, 120000),
                pollMs: number(config.pollMs, 750, 250, 5000)
            });
            this.origin = loc.origin;
            this.sessionName = 'bookmarkletFormsSessionV6:' + config.responsesId;
            this.keyPrefix = 'bookmarkletFormsBrowserV6:' + config.responsesId + ':';
            this.queue = Promise.resolve();
            this.generation = 0;
            this.active = null;
            this.timer = null;
            this.options = { heartbeatSeconds: 60, checkBeforeLaunch: false };
            this.checkJob = null;
            this.onProgress = null;
        }
        /* -- Local storage: saved session + per-user browser key -------------- */
        get storage() {
            try {
                return global.localStorage;
            }
            catch (_) {
                throw failure('STORAGE_ERROR', 'Browser storage is blocked on this site.');
            }
        }
        _serial(fn) {
            const generation = this.generation;
            const job = this.queue.catch(() => {
            }).then(() => {
                if (generation !== this.generation)
                    throw failure('CANCELLED');
                return fn();
            });
            this.queue = job;
            return job;
        }
        _saved() {
            try {
                return JSON.parse(this.storage.getItem(this.sessionName) || 'null');
            }
            catch (_) {
                return null;
            }
        }
        _browserKey(user, create = true) {
            const name = this.keyPrefix + encodeURIComponent(user.trim().toLowerCase());
            let value = this.storage.getItem(name);
            if (!/^[A-Za-z0-9_-]{43}$/.test(value || '')) {
                if (!create)
                    return '';
                value = randomKey();
                this.storage.setItem(name, value);
                if (this.storage.getItem(name) !== value)
                    throw failure('STORAGE_ERROR', 'Browser storage is unavailable.');
            }
            return value;
        }
        /* -- Request plumbing: serial queue, cancellation, progress, timeouts - */
        cancelPendingRequests() {
            this.generation++;
            if (this.active)
                this.active.cancel();
        }
        _progress(s, phase, detail) {
            if (s.closed)
                return;
            s.phase = phase;
            if (typeof this.onProgress === 'function') {
                try {
                    this.onProgress({ build: BUILD, id: s.id, phase, detail: detail || '', elapsedMs: clock() - s.start });
                }
                catch (_) {
                }
            }
        }
        // Settles even when a page-wrapped fetch ignores AbortController.
        _bounded(s, make, ms, cleanup) {
            return new Promise((resolve, reject) => {
                let done = false, timer;
                const finish = (err, value) => {
                    if (done)
                        return;
                    done = true;
                    clearTimeout(timer);
                    s.pending.delete(cancel);
                    try {
                        if (cleanup)
                            cleanup();
                    }
                    catch (_) {
                    }
                    if (err)
                        reject(err);
                    else
                        resolve(value);
                };
                const cancel = () => finish(failure('CANCELLED'));
                if (s.closed) {
                    cancel();
                    return;
                }
                s.pending.add(cancel);
                timer = setTimeout(() => finish(new Error('This operation did not finish before its timeout.')), Math.max(1, Math.min(ms, s.end - clock())));
                Promise.resolve().then(() => {
                    if (s.closed)
                        throw failure('CANCELLED');
                    return make();
                })
                    .then(v => finish(null, v), e => finish(e));
            });
        }
        /* -- Reading replies: CSV export and gviz JSONP query (raced) -------- */
        _readCSV(s) {
            const controller = new AbortController();
            const url = 'https://docs.google.com/spreadsheets/d/' + this.config.responsesId +
                '/export?format=csv&gid=' + this.config.responsesGid + '&t=' + Date.now();
            return this._bounded(s, async () => {
                const response = await global.fetch(url, { credentials: 'omit', cache: 'no-store', signal: controller.signal });
                if (!response.ok)
                    throw new Error('CSV read returned HTTP ' + response.status + '.');
                const text = await response.text();
                if (/^\s*</.test(text))
                    throw new Error('CSV read returned an HTML sign-in/error page.');
                const rows = parseCSV(text);
                if (!rows[0] || rows[0][0] !== 'request_id' || rows[0][3] !== 'response_envelope')
                    throw new Error('CSV read did not contain the expected response-sheet headers.');
                const row = rows.slice(1).find(r => r[0] === s.id);
                return row ? row[3] : null;
            }, 6000, () => controller.abort());
        }
        _readQuery(s) {
            const doc = global.document;
            if (!doc || !doc.createElement)
                return Promise.reject(new Error('No page document for the Sheets query.'));
            const name = '__blq_' + randomId(), reqId = String(++s.sequence);
            let script;
            const cleanup = () => {
                if (script) {
                    script.onload = null;
                    script.onerror = null;
                    script.remove();
                }
                try {
                    delete global[name];
                }
                catch (_) {
                    global[name] = undefined;
                }
            };
            return this._bounded(s, () => new Promise((resolve, reject) => {
                global[name] = data => {
                    try {
                        if (!data || data.status !== 'ok' || !data.table || !Array.isArray(data.table.rows))
                            throw new Error('Google Sheets query returned an access or query error.');
                        if (data.reqId !== undefined && String(data.reqId) !== reqId)
                            throw new Error('Unexpected Sheets query identifier.');
                        const row = data.table.rows.find(r => r && Array.isArray(r.c) && r.c[0] && r.c[0].v === s.id);
                        if (!row) {
                            resolve(null);
                            return;
                        }
                        if (!row.c[1] || typeof row.c[1].v !== 'string')
                            throw new Error('Missing response envelope.');
                        resolve(row.c[1].v);
                    }
                    catch (e) {
                        reject(e);
                    }
                };
                script = doc.createElement('script');
                script.async = true;
                script.referrerPolicy = 'no-referrer';
                const qs = new URLSearchParams({
                    gid: String(this.config.responsesGid), headers: '1',
                    tq: "select A, D where A = '" + s.id + "' limit 1",
                    tqx: 'version:0.6;out:json;reqId:' + reqId + ';responseHandler:' + name,
                    t: String(Date.now())
                });
                // Only Google's public response-sheet query is loaded as script.
                // No password, reply key, browser key or session token appears in this URL.
                script.src = 'https://docs.google.com/spreadsheets/d/' + this.config.responsesId + '/gviz/tq?' + qs.toString();
                script.onerror = () => reject(new Error('Google Sheets query was blocked or could not load.'));
                script.onload = () => reject(new Error('Google Sheets loaded without its expected data callback.'));
                (doc.head || doc.documentElement).appendChild(script);
            }), 6000, cleanup);
        }
        /* -- Verifying replies: HMAC signature, request id, action, expiry ---- */
        async _verify(s, envelope, key, action, sentAt) {
            this._progress(s, 'Verifying reply', 'Checking the request ID, signature, action and expiry.');
            let env, result;
            try {
                env = JSON.parse(envelope);
                if (!env || env.v !== 1 || typeof env.payload !== 'string' || env.payload.length > 45000 ||
                    typeof env.mac !== 'string' || !/^[A-Za-z0-9_-]{43}$/.test(env.mac))
                    throw failure('INVALID_REPLY');
                const valid = await this._bounded(s, () => global.crypto.subtle.verify('HMAC', key, unb64(env.mac), new TextEncoder().encode(env.payload)), 5000);
                if (!valid)
                    throw failure('INVALID_REPLY');
                result = JSON.parse(env.payload);
            }
            catch (e) {
                if (e.code === 'CANCELLED')
                    throw e;
                throw failure('INVALID_REPLY');
            }
            const now = Date.now();
            if (!result || result.v !== 6 || result.id !== s.id || result.action !== action || typeof result.ok !== 'boolean' ||
                !Number.isFinite(result.createdAt) || !Number.isFinite(result.expiresAt) ||
                result.expiresAt <= now || result.expiresAt <= result.createdAt ||
                result.createdAt < sentAt - 30000 || result.createdAt > now + 30000)
                throw failure('INVALID_REPLY');
            if (!result.ok)
                throw failure(result.code, messages[result.code] || 'The server rejected this request.');
            if (action === 'ping' && result.code !== 'READY')
                throw failure('INVALID_REPLY');
            if (['login', 'check_session'].includes(action) &&
                (result.code !== 'OK' || !Number.isFinite(result.sessionExpiresAt) || result.sessionExpiresAt <= now))
                throw failure('INVALID_REPLY');
            if (action === 'logout' && result.code !== 'LOGGED_OUT')
                throw failure('INVALID_REPLY');
            return result;
        }
        /* -- One full round trip: submit Form → poll sheet → verify reply ----- */
        async _request(action, extra) {
            const start = clock(), s = {
                id: randomId(), start, end: start + this.config.timeoutMs, closed: false,
                pending: new Set(), phase: 'Starting', readSucceeded: false, lastReadError: '', sequence: 0
            };
            let rejectGuard;
            const guard = new Promise((_, reject) => {
                rejectGuard = reject;
            });
            const dispose = () => {
                if (s.closed)
                    return;
                s.closed = true;
                for (const cancel of [...s.pending])
                    cancel();
            };
            const fail = err => {
                rejectGuard(err);
                dispose();
            };
            s.cancel = () => fail(failure('CANCELLED'));
            const totalTimer = setTimeout(() => {
                const code = s.readSucceeded ? 'TIMEOUT' : 'TRANSPORT_ERROR';
                fail(failure(code, messages[code] + ' Last step: ' + s.phase +
                    (s.lastReadError ? ' ' + s.lastReadError : '')));
            }, this.config.timeoutMs);
            this.active = s;
            const run = async () => {
                const sentAt = Date.now(), replySecret = randomKey();
                const body = Object.assign({}, extra, { v: 6, id: s.id, action, replySecret, sentAt, origin: this.origin });
                this._progress(s, 'Preparing request');
                const key = await this._bounded(s, () => global.crypto.subtle.importKey('raw', unb64(replySecret), { name: 'HMAC', hash: 'SHA-256' }, false, ['verify']), 5000);
                const params = new URLSearchParams();
                params.set(this.config.fields.requestId, s.id);
                params.set(this.config.fields.payload, JSON.stringify(body));
                const controller = new AbortController();
                this._progress(s, 'Submitting Google Form', 'This is only a submission attempt, not approval.');
                // Start the POST, but do not wait for Google's confirmation page/redirects.
                // Receiving and verifying a request-specific reply is the only success signal.
                const post = this._bounded(s, () => global.fetch(this.config.formPostUrl, {
                    method: 'POST', mode: 'no-cors', credentials: 'omit', body: params, signal: controller.signal
                }), 8000, () => controller.abort()).catch(e => {
                    if (!s.closed)
                        this._progress(s, 'Waiting for server reply', 'Submission confirmation was unavailable; awaiting the signed reply.');
                });
                const pause = async (ms) => {
                    let timer;
                    await this._bounded(s, () => new Promise(resolve => {
                        timer = setTimeout(resolve, ms);
                    }), Math.min(ms + 200, s.end - clock()), () => clearTimeout(timer));
                };
                const poll = async (mode, initialDelay) => {
                    if (initialDelay)
                        await pause(initialDelay);
                    while (!s.closed && clock() < s.end) {
                        let envelope = null;
                        try {
                            envelope = mode === 'query' ? await this._readQuery(s) : await this._readCSV(s);
                            s.readSucceeded = true;
                        }
                        catch (e) {
                            if (s.closed)
                                throw failure('CANCELLED');
                            s.lastReadError = (mode === 'query' ? 'Sheets query: ' : 'CSV: ') + e.message;
                        }
                        if (envelope !== null) {
                            const result = await this._verify(s, envelope, key, action, sentAt);
                            if (s.closed)
                                throw failure('CANCELLED');
                            this._progress(s, 'Verified reply', result.code);
                            return result;
                        }
                        this._progress(s, 'Waiting for server reply');
                        await pause(this.config.pollMs);
                    }
                    throw failure(s.closed ? 'CANCELLED' : s.readSucceeded ? 'TIMEOUT' : 'TRANSPORT_ERROR');
                };
                return await Promise.race([poll('query', 0), poll('csv', 150)]);
            };
            try {
                return await Promise.race([run(), guard]);
            }
            catch (e) {
                if (!(e instanceof AuthError)) {
                    const code = s.readSucceeded ? 'TIMEOUT' : 'TRANSPORT_ERROR';
                    e = failure(code, messages[code] + ' Last step: ' + s.phase);
                }
                e.requestId = s.id;
                throw e;
            }
            finally {
                clearTimeout(totalTimer);
                dispose();
                if (this.active === s)
                    this.active = null;
            }
        }
        /* -- Public session API: ping, resume, login, check, logout ---------- */
        ping() {
            return this._serial(() => this._request('ping', {}));
        }
        _validSession(saved) {
            return !!saved && typeof saved.username === 'string' && saved.username.trim().length > 0 &&
                /^[A-Za-z0-9_-]{43}$/.test(saved.sessionToken || '') &&
                Number.isFinite(saved.expiresAt) && saved.expiresAt > Date.now();
        }
        _catalog(games) {
            if (!games || typeof games !== 'object' || Array.isArray(games))
                return null;
            const clean = Object.create(null);
            for (const [name, g] of Object.entries(games)) {
                if (!name || !g || typeof g.url !== 'string' || !/^https:\/\//i.test(g.url))
                    return null;
                clean[name] = { url: g.url, icon: String(g.icon || ''), type: String(g.type || '') };
            }
            return clean;
        }
        resume() {
            const saved = this._saved();
            if (!this._validSession(saved) || saved.origin !== this.origin)
                return null;
            const games = this._catalog(saved.games);
            if (!games || !Number.isFinite(saved.lastVerifiedAt) ||
                saved.lastVerifiedAt > Date.now() + 30000)
                return null;
            if (saved.protected === true && !this._browserKey(saved.username, false))
                return null;
            this.options = { heartbeatSeconds: 60, checkBeforeLaunch: false };
            return {
                ok: true, code: 'OK', username: saved.username, games,
                sessionExpiresAt: saved.expiresAt, protected: saved.protected === true,
                cached: true, lastVerifiedAt: saved.lastVerifiedAt
            };
        }
        _store(saved, result) {
            const games = this._catalog(result.games) || this._catalog(saved.games);
            const value = {
                username: saved.username, sessionToken: saved.sessionToken, expiresAt: result.sessionExpiresAt,
                origin: this.origin, lastVerifiedAt: Date.now(), games,
                protected: typeof result.protected === 'boolean' ? result.protected : saved.protected === true,
                heartbeatSeconds: result.heartbeatSeconds || 60
            };
            this.storage.setItem(this.sessionName, JSON.stringify(value));
            this.options = { heartbeatSeconds: value.heartbeatSeconds, checkBeforeLaunch: false };
            return value;
        }
        login(username, password) {
            return this._serial(async () => {
                username = String(username).trim();
                const browserKey = this._browserKey(username), sessionToken = randomKey();
                const result = await this._request('login', {
                    username, password: String(password), browserKey, sessionToken, includeGames: true
                });
                if (!this._catalog(result.games))
                    throw failure('INVALID_REPLY', 'The server did not return a valid game catalog.');
                try {
                    this._store({ username, sessionToken }, result);
                }
                catch (_) {
                    throw failure('STORAGE_ERROR', 'Login was accepted but the browser could not save the session.');
                }
                return result;
            });
        }
        check(includeGames = true) {
            // One network recheck shared by heartbeat, focus and resume; no queue of duplicate checks.
            if (this.checkJob)
                return this.checkJob;
            this.checkJob = this._serial(async () => {
                const saved = this._saved();
                if (!this._validSession(saved))
                    throw failure('INVALID_SESSION');
                const browserKey = this._browserKey(saved.username, false);
                try {
                    const result = await this._request('check_session', {
                        sessionToken: saved.sessionToken, browserKey, includeGames
                    });
                    // Another tab may have signed out or signed into a different account meanwhile.
                    const current = this._saved();
                    if (!current || current.sessionToken !== saved.sessionToken)
                        throw failure('CANCELLED');
                    this._store(saved, result);
                    return result;
                }
                catch (e) {
                    if (['ACCOUNT_BANNED', 'ACCOUNT_DISABLED', 'INVALID_SESSION', 'SESSION_REVOKED', 'BROWSER_LOCKED'].includes(e.code)) {
                        const current = this._saved();
                        if (current && current.sessionToken === saved.sessionToken)
                            this.storage.removeItem(this.sessionName);
                    }
                    throw e;
                }
            }).finally(() => {
                this.checkJob = null;
            });
            return this.checkJob;
        }
        logout() {
            this.stopHeartbeat();
            const saved = this._saved();
            this.storage.removeItem(this.sessionName); // local logout is immediate
            this.cancelPendingRequests();
            return this._serial(async () => {
                if (saved && /^[A-Za-z0-9_-]{43}$/.test(saved.sessionToken || ''))
                    return this._request('logout', { sessionToken: saved.sessionToken });
                return { ok: true, code: 'LOGGED_OUT' };
            });
        }
        /* -- Heartbeat: background re-check every heartbeatSeconds (≥ 60 s) -- */
        startHeartbeat(onBlocked, onRestored) {
            this.stopHeartbeat();
            let stopped = false;
            const run = async () => {
                try {
                    const r = await this.check(true);
                    if (onRestored)
                        onRestored(r);
                }
                catch (e) {
                    if (onBlocked)
                        onBlocked(e);
                }
                if (!stopped)
                    this.timer = setTimeout(run, Math.max(60, this.options.heartbeatSeconds || 60) * 1000);
            };
            this.stopHeartbeat = () => {
                stopped = true;
                if (this.timer)
                    clearTimeout(this.timer);
                this.timer = null;
            };
            this.timer = setTimeout(run, Math.max(60, this.options.heartbeatSeconds || 60) * 1000);
        }
        stopHeartbeat() {
            if (this.timer)

    /* Full-screen overlay in a shadow root, so the host page's CSS can't leak in.
     * Used before the library page exists: sign-in, restoring, notices. */
    function mountShadow(id, html) {
        ensureFonts();
        var host = document.createElement('div');
        host.id = id;
        host.dataset.edition = THEME.id;
        host.style.cssText = 'all:initial;position:fixed;inset:0;z-index:2147483647;display:block;';
        var root = host.attachShadow({ mode: 'open' });
        root.innerHTML = '<style>' + THEME.css.shadow + '</style>' + html;
        (document.body || document.documentElement).appendChild(host);
        return { host: host, root: root };
    }

    var auth = null;      // current BookmarkletFormsAuth instance

    function showLogin(error) {
        window.__blFormsOpeningV8 = false;
        if (login && login.host.isConnected) {
            if (error) loginError(error);
            return;
        }
        login = mountShadow(HOST_ID.login, THEME.markup.login());
        var r = login.root;
        var u = byId('u', r), p = byId('p', r), btn = byId('lBtn', r), eye = byId('eyeBtn', r);
        var authBusy = false;
        u.focus();
        // No saved username or password is ever put into these fields.
        p.setAttribute('autocomplete', 'off');

        eye.onclick = function () {
            var show = p.type === 'password';
            p.type = show ? 'text' : 'password';
            eye.innerHTML = show ? ICON.eyeOff : ICON.eye;
            eye.setAttribute('aria-label', show ? 'Hide password' : 'Show password');
            p.focus();
        };
        u.onkeydown = function (e) { if (e.key === 'Enter') p.focus(); };
        p.onkeydown = function (e) { if (e.key === 'Enter') btn.click(); };

        function setLoading(on) {
            btn.disabled = on;
            btn.classList.toggle('busy', on);
            byId('lBtnTxt', r).textContent = on ? COPY.signingIn : COPY.signIn;
            u.disabled = on;
            p.disabled = on;
            setPhase(on ? COPY.phase['Preparing request'] || '' : '');
        }
        function setPhase(text) {
            var el = byId('lPhase', r);
            el.textContent = text;
            el.classList.toggle('on', !!text);
        }

        btn.onclick = async function () {
            if (authBusy) return;
            var username = u.value.trim();
            var password = p.value;
            if (!username || !password) {
                loginError({ title: COPY.missingFields });
                (username ? p : u).focus();
                return;
            }
            authBusy = true;
            setLoading(true);
            hideLoginError();
            auth.onProgress = function (ev) {
                if (login) setPhase(COPY.phase[ev.phase] || ev.phase);
            };
            try {
                var approved = await auth.login(username, password);
                auth.onProgress = null;
                closeLogin();
                loadLibrary(approved.games, username, Date.now(), false);
            } catch (e) {
                if (login) {
                    loginError(e);
                } else {
                    console.error('Library render failed', e && e.stack || e);
                    showNotice(COPY.openFailedTitle, (e && e.message) || String(e));
                }
            } finally {
                auth.onProgress = null;
                password = '';
                if (login) {
                    p.value = '';
                    setLoading(false);
                    (u.value.trim() ? p : u).focus();
                }
                authBusy = false;
            }
        };
        if (error) loginError(error);
    }

    function loginError(e) {

        box.innerHTML = errorHTML(d);
        // Restart the shake animation on repeat errors.
        void box.offsetWidth;
        box.classList.add('shake');
    }
    function hideLoginError() {
        if (login) byId('lMsg', login.root).className = 'emsg';
    }

    /* ##########################################################################
     * §7  SCREEN · LIBRARY
     * --------------------------------------------------------------------------
     *  loadLibrary() replaces the page with the library and owns everything
     *  that lives on it. Its parts:
     *     7a  local data      playtime, favorites, recents, avatar, prefs
     *     7b  page + header   write the page, fill in names and counts
     *     7c  profile panel   stats, playtime, achievements, appearance
     *     7d  game grid       cards, sections, search
     *  §8  access control and §9 game loader are also inside, further down.
     * ########################################################################## */
    function loadLibrary(games, userName, startTime, resumed) {
        var gCount = Object.keys(games).length;
        var K = storageKeys(userName);

        /* ---- 7a. Local data ---------------------------------------------------- */
        function readJSON(key, fallback) {
            try { var v = JSON.parse(localStorage.getItem(key)); return v == null ? fallback : v; }
            catch (_) { return fallback; }
        }
        function writeJSON(key, value) {
            try { localStorage.setItem(key, JSON.stringify(value)); } catch (_) { }
        }
        if (!localStorage.getItem(K.member)) localStorage.setItem(K.member, Date.now().toString());

        // Playtime that an instrumented game reported while the library was closed
        // (ignored if older than two hours).
        (function () {
            var d = readJSON(TIME_DELTA_KEY, null);
            if (!d) return;
            if (d.name && d.ms && (Date.now() - (d.ts || 0)) < 2 * 60 * 60 * 1000) {
                var s = getStats();
                s[d.name] = (s[d.name] || 0) + d.ms;
                writeJSON(K.stats, s);
            }
            localStorage.removeItem(TIME_DELTA_KEY);
        })();

        function getStats() { var s = readJSON(K.stats, {}); return s && typeof s === 'object' ? s : {}; }
        function addPlaytime(name, ms) {
            if (!name || ms < 500) return;
            var s = getStats();
            s[name] = (s[name] || 0) + ms;
            writeJSON(K.stats, s);
        }
        function getFavorites() { var f = readJSON(K.favorites, []); return Array.isArray(f) ? f : []; }
        function isFavorite(n) { return getFavorites().indexOf(n) > -1; }
        function toggleFavorite(n) {
            var f = getFavorites(), i = f.indexOf(n);
            if (i > -1) f.splice(i, 1); else f.unshift(n);
            writeJSON(K.favorites, f);
        }
        function getRecent() { var r = readJSON(K.recent, []); return Array.isArray(r) ? r : []; }
        function pushRecent(n) {
            var r = getRecent().filter(function (x) { return x !== n; });
            r.unshift(n);
            writeJSON(K.recent, r.slice(0, 10));
        }
        function totalPlaytime() {
            return Object.values(getStats()).reduce(function (a, b) { return a + b; }, 0);
        }
        function topGame() {
            var s = getStats(), k = Object.keys(s);
            return k.length ? k.reduce(function (a, b) { return s[a] > s[b] ? a : b; }) : null;
        }
        function formatDuration(ms) {
            var s = Math.floor(ms / 1000), h = Math.floor(s / 3600), m = Math.floor((s % 3600) / 60), sc = s % 60;
            if (h > 0) return h + 'h ' + m + 'm';
            if (m > 0) return m + 'm ' + sc + 's';
            return sc > 0 ? sc + 's' : '<1s';
        }

        // Achievement rules are shared; their names come from the theme's copy.
        function achievements() {
            var s = getStats(), tot = totalPlaytime(), fv = getFavorites(), rc = getRecent();
            var played = Object.keys(s).length;
            var rules = [
                { icon: ICON.controller, u: rc.length >= 1 },
                { icon: ICON.controller, u: played >= 3 },
                { icon: ICON.controller, u: played >= 5 },
                { icon: ICON.clock, u: tot >= 3600000 },
                { icon: ICON.clock, u: tot >= 18000000 },
                { icon: ICON.heart, u: fv.length >= 3 },
                { icon: ICON.heart, u: fv.length >= 5 },
                { icon: ICON.star, u: rc.length >= 10 }
            ];
            return rules.map(function (r, i) {
                return { icon: r.icon, u: r.u, name: COPY.achievements[i][0], desc: COPY.achievements[i][1] };
            });
        }
        function announceAchievements(prev) {
            achievements().forEach(function (a, i) {
                if (a.u && !prev[i].u) setTimeout(function () { toast(COPY.achievementToast(a.name), ICON.trophy, 'gold'); }, 400);
            });
        }

        /* ---- 7b. Page, header and hero ------------------------------------------ */
        writeLibraryPage(THEME.markup.library({ user: esc(userName), count: gCount }));

        function paintCounts() {
            byId('rCount').textContent = COPY.resourceCount(gCount);
            byId('ws').textContent = COPY.lede(gCount);
            byId('ftYr').textContent = COPY.footerRight(gCount);
            byId('sb').placeholder = COPY.searchPlaceholder(gCount);
        }
        paintCounts();
        byId('sDate').textContent = COPY.dateValue(new Date());
        byId('uName').textContent = userName;
        byId('phName').textContent = userName;
        byId('wg').innerHTML = COPY.greeting(new Date().getHours(), esc(userName));
        var memberSince = new Date(parseInt(localStorage.getItem(K.member), 10))
            .toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
        byId('phSub').textContent = COPY.memberSince(memberSince);

        /* Status pill / cell: online, reconnecting, paused, banned, signin. */
        function setAccessState(state) {
            var el = byId('accessStatus'), txt = byId('accessText');
            if (el) el.dataset.state = state;
            if (txt) txt.textContent = COPY.status[state] || state;
        }
        setAccessState('online');

        /* Appearance (theme variants) and the optional effects switch. */
        function applyAppearance(id) {
            var ids = THEME.appearances.map(function (a) { return a.id; });
            if (ids.indexOf(id) < 0) id = ids[0];
            ids.forEach(function (a) { document.body.classList.toggle('th-' + a, a === id); });
            try { localStorage.setItem(K.appearance, id); } catch (_) { }
            document.querySelectorAll('.tbtn[data-theme]').forEach(function (b) {
                var on = b.dataset.theme === id;
                b.classList.toggle('active', on);
                b.setAttribute('aria-pressed', String(on));
            });
        }
        function applyEffects(on) {
            document.body.classList.toggle('fx-off', !on);
            try { localStorage.setItem(K.effects, on ? 'on' : 'off'); } catch (_) { }
            document.querySelectorAll('.tbtn[data-fx]').forEach(function (b) {
                var active = (b.dataset.fx === 'on') === on;
                b.classList.toggle('active', active);
                b.setAttribute('aria-pressed', String(active));
            });
        }
        applyAppearance(localStorage.getItem(K.appearance) || THEME.appearances[0].id);

                    if (w > h) { if (w > max) { h = h * (max / w); w = max; } }

                };

                img.src = reader.result;
            };
            reader.readAsDataURL(f);
        };

        /* ##########################################################################
         * §8  LOGIC · ACCESS CONTROL  (bans, revocations, reconnects, sign-out)
         * --------------------------------------------------------------------------
         *  The library opens from the last accepted session and re-checks in the
         *  background, so launching a game never waits on Google Forms.
         *  A confirmed ban/revocation closes running games. Enforcement is
         *  eventual (next heartbeat / focus), not instantaneous.
         * ########################################################################## */
        var launchBusy = false, accessBlocked = false, disposed = false, gameGeneration = 0;
        var gameWindows = new Set(), gameControllers = new Set(), accessCheck = null;
        var terminalAccessError = null;
        var openedSession = auth._saved(), lastVerified = openedSession && openedSession.lastVerifiedAt || Date.now();
        var sessionTokenAtOpen = openedSession && openedSession.sessionToken;
        var expiryTimer = null, focusHandler = null, storageHandler = null;

        function closeRunningGames() {
            gameGeneration++;
            saveCurrentTime();
            curGame = null;
            writeGameOpen = false;
            gameControllers.forEach(function (c) { c.abort(); });
            gameControllers.clear();
            if (frame) {
                frame.srcdoc = '';
                frame.removeAttribute('src');

            gameWindows.clear();
        }

        function acceptAccess(result) {
            if (disposed || terminalAccessError) return;
            accessBlocked = false;
            var cached = auth._saved();
            lastVerified = cached && cached.lastVerifiedAt || Date.now();
            var panel = byId(HOST_ID.block);
            if (panel) panel.remove();
            setAccessState('online');
            // The owner changed the catalog: re-render with the new list.
            if (result && result.games && JSON.stringify(result.games) !== JSON.stringify(games)) {
                games = result.games;
                gCount = Object.keys(games).length;
                paintCounts();
                renderFavorites(false);
                renderRecent(false);
                renderAll(false);
                doFilter();
            }
        }

        function disposeLibrary() {
            if (disposed) return;
            disposed = true;
            auth.stopHeartbeat();
            auth.cancelPendingRequests();
            clearInterval(deltaTimer);
            clearInterval(expiryTimer);
            window.removeEventListener('beforeunload', saveCurrentTime);
            if (focusHandler) window.removeEventListener('focus', focusHandler);
            if (storageHandler) window.removeEventListener('storage', storageHandler);
            if (window.__blLibraryV8) window.__blLibraryV8 = null;
        }

        function returnToSignIn(reason) {
            closeRunningGames();
            disposeLibrary();
            try { auth.storage.removeItem(auth.sessionName); } catch (_) { }
            var panel = byId(HOST_ID.block);
            if (panel) panel.remove();
            delete document.body.dataset.formsLibrary;
            auth = new window.BookmarkletFormsAuth(auth.config);
            showLogin(reason);
        }

        /* Locks the library behind a dialog.
         *   banned    → "Account banned", no retry (ban notices never offer one)
         *   permanent → "Sign-in required" (revoked, expired, browser-locked …)
         *   otherwise → "Access paused" with "Retry connection" (network trouble) */
        function blockAccess(error) {
            if (disposed || terminalAccessError) return;
            var code = error && error.code;
            var banned = code === 'ACCOUNT_BANNED';
            var permanent = DENY_CODES.indexOf(code) >= 0;
            if (permanent) {
                terminalAccessError = error;
                // Stop the heartbeat so a later INVALID_SESSION can't replace the
                // ban notice after the auth client removes the rejected session.
                auth.stopHeartbeat();
            }
            accessBlocked = true;
            closeRunningGames();
            closePanel();
            var kind = banned ? 'banned' : permanent ? 'signin' : 'paused';
            setAccessState(kind);
            var A = COPY.access[kind], actions = [];
            if (!permanent) {
                actions.push({
                    label: COPY.access.retry, kind: 'primary', keepOpen: true,
                    onClick: function (btn) {
                        btn.disabled = true;
                        btn.textContent = COPY.access.retrying;
                        verifyAccess().finally(function () {
                            btn.disabled = false;
                            btn.textContent = COPY.access.retry;
                        });
                    }
                });
            }
            actions.push({
                label: banned ? COPY.access.useAnother : COPY.access.signIn,
                kind: permanent ? 'primary' : 'ghost',
                onClick: function () { returnToSignIn(); }
            });
            // Themes with friendlyMessages show their own sentence and keep the
            // server's technical text as a small detail line.
            var serverText = error && error.message || '';
            var friendly = !!COPY.access.friendlyMessages;
            openDialog({
                id: HOST_ID.block,
                tone: banned ? 'danger' : permanent ? 'warn' : 'info',
                label: A.label,
                title: A.title,
                message: banned || friendly ? A.message : serverText || A.message,
                detail: !banned && friendly ? serverText : '',
                code: banned ? '' : code,
                actions: actions
            });
        }

        function failedAccess(error) {
            if (disposed || terminalAccessError || error && error.code === 'CANCELLED') return;
            if (DENY_CODES.indexOf(error && error.code) >= 0) {
                blockAccess(error);
                return;
            }
            // Network failures never erase the session or turn into "wrong password".
            // A long outage pauses play but keeps the saved session for a retry.
            setAccessState('reconnecting');
            if (Date.now() - lastVerified > TIMING.offlineGraceMs) blockAccess(error);
        }

        function verifyAccess() {
            if (disposed || terminalAccessError) return Promise.resolve(null);
            if (accessCheck) return accessCheck;
            accessCheck = auth.check(true)
                .then(function (r) { acceptAccess(r); return r; })
                .catch(function (e) { failedAccess(e); return null; })
                .finally(function () { accessCheck = null; });
            return accessCheck;
        }

        function canLaunch() {
            if (disposed || accessBlocked) return false;
            var saved = auth._saved();
            if (!auth._validSession(saved) || saved.sessionToken !== sessionTokenAtOpen) {
                blockAccess(Object.assign(new Error('Your saved session expired or was signed out.'), { code: 'INVALID_SESSION' }));
                return false;
            }
            return true;
        }

        function signOutLibrary() {
            var previousAuth = auth;
            closeRunningGames();
            disposeLibrary();
            // Local sign-out is instant; the server-side revoke runs in the background.
            previousAuth.logout().catch(function () { });
            delete document.body.dataset.formsLibrary;
            auth = new window.BookmarkletFormsAuth(previousAuth.config);
            showLogin();
        }

        /* ##########################################################################
         * §9  LOGIC · GAME LOADER  (download + cache, in-page overlay, pop-outs)
         * ########################################################################## */
        var assetCache = new Map(), assetPending = new Map(), assetBytes = 0;

        function fetchGame(gd) {
            var url = String(gd.url || '');
            if (!/^https:\/\//i.test(url)) return Promise.reject(new Error('The game URL must use HTTPS.'));
            if (assetCache.has(url)) return Promise.resolve(assetCache.get(url));
            if (assetPending.has(url)) return assetPending.get(url);
            var controller = new AbortController(), timer;
            gameControllers.add(controller);
            var job = Promise.race([
                (async function () {
                    var res = await fetch(url, { signal: controller.signal });
                    if (!res.ok) throw new Error('Game download failed (HTTP ' + res.status + ').');
                    return await res.text();
                })(),
                new Promise(function (_, reject) {
                    timer = setTimeout(function () {
                        controller.abort();
                        reject(new Error('Game download timed out.'));
                    }, TIMING.gameDownloadMs);
                })
            ]).then(function (html) {
                // Cache small HTML launchers only, never huge embedded game bundles.
                var size = html.length * 2;
                if (size <= GAME_CACHE.maxItemBytes) {
                    while (assetBytes + size > GAME_CACHE.maxTotalBytes && assetCache.size) {
                        var oldest = assetCache.keys().next().value;
                        assetBytes -= assetCache.get(oldest).length * 2;
                        assetCache.delete(oldest);
                    }
                    if (!assetCache.has(url)) {
                        assetCache.set(url, html);
                        assetBytes += size;
                    }
                }
                return html;
            }).finally(function () {
                clearTimeout(timer);
                gameControllers.delete(controller);
                assetPending.delete(url);
            });
            assetPending.set(url, job);
            return job;
        }

        function prefetchGame(gd) {
            if (disposed || accessBlocked || assetPending.size >= 2) return;
            fetchGame(gd).catch(function () { });
        }

        // "iframe" games on google.* pages pop out to the site's own origin first.
        function waitForPopup(nw, googleHost) {
            if (!googleHost) return Promise.resolve();
            return new Promise(function (resolve, reject) {
                var until = Date.now() + TIMING.popupHostWaitMs;
                (function ready() {
                    if (nw.closed) { reject(new Error('The game window was closed.')); return; }
                    try {
                        if (nw.location.href !== 'about:blank' && nw.document.readyState === 'complete') { resolve(); return; }
                    } catch (_) { }
                    if (Date.now() >= until) { reject(new Error('The game host page did not finish loading.')); return; }
                    setTimeout(ready, 50);
                })();
            });
        }

        }

        async function launchGame(name, gd) {
            if (launchBusy || !canLaunch()) return;
            launchBusy = true;
            setCardLoading(name, true);
            var nw = null, mounted = false, generation = ++gameGeneration;
            var separate = gd.type === 'write' || gd.type === 'iframe';
            var googleHost = gd.type === 'iframe' && /(^|\.)google\./i.test(location.hostname);
            if (separate) {
                // Open during the click so pop-up blockers allow it.
                nw = window.open(googleHost ? location.origin + '/' : 'about:blank', '_blank');
                if (!nw) {
                    launchBusy = false;
                    setCardLoading(name, false);
                    showPopupBlocked(name, gd);
                    return;
                }
                gameWindows.add(nw);
                if (!googleHost) paintPopupLoading(nw, name);
            }
            try {
                // No Form request or session round trip here.
                var values = await Promise.all([fetchGame(gd), nw ? waitForPopup(nw, googleHost) : Promise.resolve()]);
                var h = values[0];
                if (disposed || accessBlocked || generation !== gameGeneration || nw && nw.closed) return;
                var prev = achievements();
                pushRecent(name);
                renderRecent(false);
                if (separate) {
                    mountPopupGame(nw, name, gd, h);
                    mounted = true;
                } else {
                    mountOverlayGame(name, h);
                }
                announceAchievements(prev);
            } catch (e) {
                if (!disposed && !accessBlocked) launchFailure(e, name, gd);
            } finally {
                launchBusy = false;
                setCardLoading(name, false);
                if (nw && !mounted) {
                    gameWindows.delete(nw);
                    try { nw.close(); } catch (_) { }
                }
            }
        }

        // type "write"  → the game HTML becomes the pop-out document.
        // type "iframe" → the pop-out hosts the game in a full-window iframe.
        function mountPopupGame(nw, name, gd, h) {
            writeGameOpen = true;
            var t0 = Date.now(), opener = window, recorded = false;
            function recordPopupTime() {
                if (recorded) return;
                recorded = true;
                try { addPlaytime(name, Date.now() - t0); } catch (_) { }
                gameWindows.delete(nw);
            }
            var doc = nw.document;
            if (gd.type === 'write') {
                doc.open();
                doc.write(h);
                doc.title = name;
                doc.close();
            } else {
                doc.title = name;
                if (doc.body) doc.body.className = '';
                doc.documentElement.style.cssText = 'margin:0;width:100%;height:100%;overflow:hidden;background:#000';
                doc.body.textContent = '';
                doc.body.style.cssText = 'margin:0;width:100%;height:100%;overflow:hidden;background:#000';
                var gameFrame = doc.createElement('iframe');
                gameFrame.style.cssText = 'position:fixed;inset:0;width:100%;height:100%;border:0;background:#000;opacity:1';
                gameFrame.allow = 'autoplay; fullscreen; gamepad';
                gameFrame.allowFullscreen = true;
                gameFrame.onload = function () {
                    setTimeout(function () { try { gameFrame.contentWindow.focus(); } catch (_) { } }, 100);
                };
                gameFrame.srcdoc = h;
                doc.body.appendChild(gameFrame);
            }
            (function addClose() {
                if (nw.closed) return;
                try {
                    if (!nw.document.body) { setTimeout(addClose, 50); return; }
                    addPopupCloseButton(nw, function () {
                        recordPopupTime();
                        try { opener.focus(); } catch (_) { }
                        nw.close();
                    });
                    nw.addEventListener('beforeunload', recordPopupTime);
                } catch (_) { }
            })();
            // If the library tab closes, the pop-out closes too.
            nw.setInterval(function () {
                try { if (opener.closed) { recordPopupTime(); nw.close(); } } catch (_) { }
            }, 5000);
        }

        function mountOverlayGame(name, h) {
            saveCurrentTime();
            curGame = name;
            curStart = Date.now();
            document.body.style.overflow = 'hidden';
            overlay.style.display = 'flex';
            setNowPlaying(name);
            requestAnimationFrame(function () { overlay.classList.add('vis'); });
            frame.classList.remove('ld');
            frame.onload = function () {
                frame.classList.add('ld');
                setTimeout(function () { try { frame.contentWindow.focus(); } catch (_) { } }, 100);
            };
            frame.srcdoc = h;
        }

        /* ##########################################################################
         * §10 LIBRARY STARTUP  (first render, heartbeat, focus/storage listeners)
         * ########################################################################## */
        renderFavorites(true);
        renderRecent(true);
        renderAll(true);
        document.body.dataset.formsLibrary = '8';
        window.__blFormsOpeningV8 = false;
        auth.onProgress = null;
        auth.startHeartbeat(failedAccess, acceptAccess);

        focusHandler = function () {
            if (document.visibilityState !== 'hidden' && Date.now() - lastVerified > TIMING.reverifyAfterMs) verifyAccess();
        };
        // Signing out (or into another account) in another tab closes this one.
        storageHandler = function (e) {
            if (e.key !== auth.sessionName) return;
            var current = auth._saved();
            if (!current || current.sessionToken !== sessionTokenAtOpen) {
                returnToSignIn(Object.assign(new Error('This session was signed out or changed in another tab.'), { code: 'INVALID_SESSION' }));
            }
        };
        window.addEventListener('focus', focusHandler);
        window.addEventListener('storage', storageHandler);
        expiryTimer = setInterval(function () {
            if (!canLaunch()) return;
            if (document.visibilityState !== 'hidden' && Date.now() - lastVerified > TIMING.offlineGraceMs && !accessCheck) verifyAccess();
        }, 60000);

        // Handle used when the bookmark is clicked again on this page.
        window.__blLibraryV8 = {
            theme: THEME.id,
            refreshIfDue: function () {
                if (Date.now() - lastVerified > TIMING.reverifyAfterMs) verifyAccess();
            },
            dispose: function () {
                closeRunningGames();
                disposeLibrary();
            }
        };

        // A cached library is shown first; this background check then confirms it.
        if (resumed) setTimeout(function () { verifyAccess(); }, 0);
    }

    /* ##########################################################################
     * §11 BOOT  (what happens when the bookmark is clicked)
     * ########################################################################## */
    function start() {
        try {
            var cached = auth.resume();
            if (cached) {
                window.__blFormsOpeningV8 = false;
                loadLibrary(cached.games, cached.username, Date.now(), true);
                return;
            }
            var saved = auth._saved();
            if (auth._validSession(saved)) {
                // One-time migration for v6/v7 tokens that had no cached game list.
                // This is a session check, not a login or password resubmission.
                showRestoring();
                auth.check(true).then(function (r) {
                    hideRestoring();
                    window.__blFormsOpeningV8 = false;
                    loadLibrary(r.games, saved.username, Date.now(), false);
                }).catch(function (e) {
                    hideRestoring();
                    showLogin(e);
                });
                return;
            }
            window.__blFormsOpeningV8 = false;
            showLogin();
        } catch (e) {
            window.__blFormsOpeningV8 = false;
            showLogin(e);
        }
    }

    function boot() {
        // 1. Library already open on this page.
        var running = window.__blLibraryV8;
        if (running && document.body && document.body.dataset.formsLibrary === '8') {
            if (running.theme === THEME.id) {
                running.refreshIfDue();
                return;
            }
            // An older build (e.g. v8.1) is open: close it and reopen this one
            // from the saved session (no password needed).
            try { running.dispose(); } catch (_) { }
        }
        // 2. A v7 library from before the update is still open.
        if (window.__blLibraryV7 && document.body && document.body.dataset.formsLibrary === '7') {
            showNotice(COPY.notices.refreshTitle, COPY.notices.refreshMessage);
            return;
        }
        // 3. Sign-in screen already showing.
        var existing = document.getElementById(HOST_ID.login);
        if (existing) {
            if (existing.dataset.edition === THEME.id) {
                try { existing.shadowRoot.getElementById('u').focus(); } catch (_) { }
                return;
            }
            existing.remove();
        }
        // 4. Still opening from a click a moment ago.
        if (window.__blFormsOpeningV8) return;
        window.__blFormsOpeningV8 = true;
        try {
            auth = new window.BookmarkletFormsAuth(BACKEND);
        } catch (e) {
            window.__blFormsOpeningV8 = false;
            var d = describeError(e);
            showNotice(d.title, d.message);
            return;
        }
        start();
    }

    boot();
})().catch(function (e) {
    window.__blFormsOpeningV8 = false;
    alert('Game Library: ' + (e && e.message || e));
});
