    const h = Math.min(Math.max(candidate.h, MIN_H), Math.max(MIN_H, innerHeight));
    const x = Math.min(Math.max(candidate.x, 44 - w), Math.max(0, innerWidth - 64));
    const y = Math.min(Math.max(candidate.y, 0), Math.max(0, innerHeight - 42));
    return { x, y, w, h };
  }

  const MAX_CONVERSATIONS = 40;
  const MAX_STORED_THOUGHT = 4000;

  let conversations = [];
  let activeId = "";
  let persistTimer = 0;

  function sanitizeNode(raw, fallbackParent) {
    return {
      id: typeof raw.id === "string" ? raw.id : uid(),
      parentId: typeof raw.parentId === "string" ? raw.parentId : fallbackParent,
      children: Array.isArray(raw.children) ? raw.children.filter(entry => typeof entry === "string") : [],
      role: raw.role === "assistant" ? "assistant" : "user",
      text: typeof raw.text === "string" ? raw.text : "",
      thought: typeof raw.thought === "string" ? raw.thought : "",
      note: typeof raw.note === "string" ? raw.note : "",
      modelLabel: typeof raw.modelLabel === "string" ? raw.modelLabel : "",
      effortLabel: typeof raw.effortLabel === "string" ? raw.effortLabel : "",
      state: raw.state && typeof raw.state === "object" && !Array.isArray(raw.state) ? raw.state : null
    };
  }

  function emptyConversation() {
    return { id: uid(), title: "New chat", createdAt: Date.now(), updatedAt: Date.now(), draft: "", nodes: {}, rootIds: [], activeLeafId: "" };
  }

  function nodePath(conversation, leafId) {
    const path = [];
    const seen = new Set();
    let id = leafId;
    while (id && conversation.nodes[id] && !seen.has(id)) {
      seen.add(id);
      path.push(conversation.nodes[id]);
      id = conversation.nodes[id].parentId;
    }
    return path.reverse();
  }

  function currentPath(conversation) {
    return conversation ? nodePath(conversation, conversation.activeLeafId) : [];
  }

  function siblingIds(conversation, node) {
    if (!node) return [];
    return node.parentId ? conversation.nodes[node.parentId]?.children || [] : conversation.rootIds;
  }

  function deepestLeaf(conversation, id) {
    let node = conversation.nodes[id];
    const seen = new Set();
    while (node && node.children.length && !seen.has(node.id)) {
      seen.add(node.id);
      node = conversation.nodes[node.children[node.children.length - 1]] || null;
    }
    return node ? node.id : "";
  }

  function addNode(conversation, parentId, data) {
    const node = {
      id: uid(),
      parentId: parentId || null,
      children: [],
      thought: "",
      note: "",
      modelLabel: "",
      effortLabel: "",
      state: null,
      ...data
    };
    conversation.nodes[node.id] = node;
    if (node.parentId && conversation.nodes[node.parentId]) conversation.nodes[node.parentId].children.push(node.id);
    else {
      node.parentId = null;
      conversation.rootIds.push(node.id);
    }
    conversation.activeLeafId = node.id;
    return node;
  }

  function dropNode(conversation, id) {
    const node = conversation.nodes[id];
    if (!node) return;
    for (const childId of node.children.slice()) dropNode(conversation, childId);
    const siblings = node.parentId ? conversation.nodes[node.parentId]?.children : conversation.rootIds;
    if (siblings) {
      const index = siblings.indexOf(id);
      if (index >= 0) siblings.splice(index, 1);
    }
    delete conversation.nodes[id];
    if (conversation.activeLeafId === id) conversation.activeLeafId = node.parentId || "";
  }

  function switchBranch(conversation, node, delta) {
    const siblings = siblingIds(conversation, node);
    const target = siblings[siblings.indexOf(node.id) + delta];
    if (!target) return;
    conversation.activeLeafId = deepestLeaf(conversation, target) || target;
    persist();
    renderThread();
  }

  function branchSwitcher(conversation, node) {
    if (!conversation || !node) return null;
    const siblings = siblingIds(conversation, node);
    if (siblings.length < 2) return null;
    const index = siblings.indexOf(node.id);

    const wrap = document.createElement("span");
    wrap.className = "branch";

    const back = document.createElement("button");
    back.type = "button";
    back.className = "bstep";
    back.innerHTML = ICONS.cleft;
    back.setAttribute("aria-label", "Previous version");
    back.disabled = index <= 0;
    back.onclick = () => switchBranch(conversation, node, -1);

    const count = document.createElement("span");
    count.className = "bcount";
    count.textContent = `${index + 1}/${siblings.length}`;

    const forward = document.createElement("button");
    forward.type = "button";
    forward.className = "bstep";
    forward.innerHTML = ICONS.cright;
    forward.setAttribute("aria-label", "Next version");
    forward.disabled = index >= siblings.length - 1;
    forward.onclick = () => switchBranch(conversation, node, 1);

    wrap.append(back, count, forward);
    return wrap;
  }

  function sanitizeConversations(raw) {
    if (!Array.isArray(raw)) return [];
    const out = [];

    for (const item of raw.slice(0, MAX_CONVERSATIONS)) {
      if (!item || typeof item !== "object") continue;
      const conversation = emptyConversation();
      if (typeof item.id === "string") conversation.id = item.id;
      if (typeof item.title === "string") conversation.title = item.title;
      conversation.createdAt = Number(item.createdAt) || Date.now();
      conversation.updatedAt = Number(item.updatedAt) || Date.now();
      conversation.draft = typeof item.draft === "string" ? item.draft : "";
      conversation.titleLocked = item.titleLocked === true;

      if (item.nodes && typeof item.nodes === "object" && !Array.isArray(item.nodes)) {
        for (const [key, value] of Object.entries(item.nodes)) {
          if (!value || typeof value !== "object") continue;
          const node = sanitizeNode(value, null);
          node.id = typeof value.id === "string" ? value.id : key;
          conversation.nodes[node.id] = node;
        }
        for (const node of Object.values(conversation.nodes)) {
          node.children = node.children.filter(childId => conversation.nodes[childId]);
          if (node.parentId && !conversation.nodes[node.parentId]) node.parentId = null;
        }
        conversation.rootIds = Object.values(conversation.nodes).filter(node => !node.parentId).map(node => node.id);
        conversation.activeLeafId = conversation.nodes[item.activeLeafId]
          ? item.activeLeafId
          : deepestLeaf(conversation, conversation.rootIds[conversation.rootIds.length - 1] || "");
      } else if (Array.isArray(item.turns)) {
        let parentId = null;
        for (const turn of item.turns) {
          if (!turn || typeof turn !== "object" || typeof turn.text !== "string") continue;
          if (turn.role !== "user" && turn.role !== "assistant") continue;
          const node = sanitizeNode(turn, parentId);
          node.children = [];
          conversation.nodes[node.id] = node;
          if (parentId) conversation.nodes[parentId].children.push(node.id);
          else conversation.rootIds.push(node.id);
          parentId = node.id;
        }
        conversation.activeLeafId = parentId || "";
      }

      out.push(conversation);
    }

    return out;
  }

  let prunedThisSession = "";

  function reportPrune(message) {
    if (prunedThisSession === message) return;
    prunedThisSession = message;
    toast(message);
  }

  function pruneBranches(conversation) {
    const keep = new Set(currentPath(conversation).map(node => node.id));
    let removed = 0;
    for (const id of Object.keys(conversation.nodes)) {
      if (keep.has(id)) continue;
      delete conversation.nodes[id];
      removed += 1;
    }
    if (!removed) return 0;
    conversation.rootIds = conversation.rootIds.filter(id => keep.has(id));
    for (const node of Object.values(conversation.nodes)) node.children = node.children.filter(id => keep.has(id));
    return removed;
  }

  function writeConversations() {
    if (!store.usable) return false;
    const commit = () => store.set("conversations", conversations.slice(0, MAX_CONVERSATIONS)) && store.set("activeId", activeId);
    if (commit()) return true;

    let dropped = 0;
    for (const conversation of conversations.slice(3)) {
      for (const node of Object.values(conversation.nodes)) {
        if (!node.thought) continue;
        node.thought = "";
        dropped += 1;
      }
    }
    if (dropped && commit()) {
      reportPrune("Reasoning from older chats was dropped to free space");
      return true;
    }

    let pruned = 0;
    for (const conversation of conversations) {
      if (conversation.id === activeId) continue;
      pruned += pruneBranches(conversation);
    }
    if (pruned && commit()) {
      reportPrune("Unused branches in older chats were pruned to free space");
      renderThread();
      return true;
    }

    while (conversations.length > 1) {
      let index = conversations.length - 1;
      while (index > 0 && conversations[index].id === activeId) index -= 1;
      if (conversations[index].id === activeId) break;
      conversations.splice(index, 1);
      if (commit()) {
        reportPrune("The oldest chats were removed to free space");
        renderRail();
        return true;
      }
    }

    reportPrune("Storage is full, so this chat is not being saved");
    return false;
  }

  function persist() {
    clearTimeout(persistTimer);
    persistTimer = setTimeout(writeConversations, 220);
  }

  function activeConversation() {
    return conversations.find(item => item.id === activeId) || null;
  }

  function touchConversation(conversation) {
    conversation.updatedAt = Date.now();
    conversations = [conversation, ...conversations.filter(item => item.id !== conversation.id)].slice(0, MAX_CONVERSATIONS);
    persist();
    renderRail();
  }

  function titleFrom(text) {
    const clean = String(text).replace(/\s+/g, " ").trim();
    if (!clean) return "New chat";
    return clean.length > 42 ? `${clean.slice(0, 42).trimEnd()}…` : clean;
  }

  function newConversation(select = true) {
    const conversation = emptyConversation();
    conversations.unshift(conversation);
    conversations = conversations.slice(0, MAX_CONVERSATIONS);
    if (select) {
      activeId = conversation.id;
      renderThread({ restoreDraft: true });
    }
    persist();
    renderRail();
    return conversation;
  }

  function selectConversation(id) {
    if (id === activeId) return;
    const current = activeConversation();
    if (current) current.draft = input.value;
    activeId = id;
    renderThread({ restoreDraft: true });
    renderRail();
    persist();
    if (compact) toggleRail(false);
    input.focus();
  }

  function deleteConversation(id) {
    const index = conversations.findIndex(item => item.id === id);
    if (index < 0) return;
    const [removed] = conversations.splice(index, 1);
    const wasActive = activeId === removed.id;
    let replacement = null;

    if (wasActive) {
      if (!conversations.length) replacement = newConversation(true);
      else {
        activeId = conversations[Math.min(index, conversations.length - 1)].id;
        renderThread({ restoreDraft: true });
      }
    }
    persist();
    renderRail();

    toast(`Deleted "${removed.title}"`, {
      label: "Undo",
      run: () => {
        if (conversations.some(item => item.id === removed.id)) return;
        if (replacement && !Object.keys(replacement.nodes).length) {
          const spare = conversations.findIndex(item => item.id === replacement.id);
          if (spare >= 0) conversations.splice(spare, 1);
        }
        conversations.splice(Math.min(index, conversations.length), 0, removed);
        if (wasActive) {
          activeId = removed.id;
          renderThread({ restoreDraft: true });
        }
        persist();
        renderRail();
      }
    });
  }

  function renderRail() {
    railList.replaceChildren();
    if (!conversations.length) {
      const blank = document.createElement("div");
      blank.className = "railempty";
      blank.textContent = "No chats yet. Start one below.";
      railList.appendChild(blank);
      return;
    }
    for (const conversation of conversations) {
      const item = document.createElement("div");
      item.className = `chatitem${conversation.id === activeId ? " active" : ""}`;

      const open = document.createElement("button");
      open.type = "button";
      open.className = "chatname";
      open.textContent = conversation.title;
      open.title = `${conversation.title} — double-click to rename`;
      open.onclick = () => selectConversation(conversation.id);
      open.ondblclick = event => {
        event.preventDefault();
        beginRename(conversation, open);
      };

      const remove = document.createElement("button");
      remove.type = "button";
      remove.className = "chatdel";
      remove.setAttribute("aria-label", `Delete ${conversation.title}`);
      remove.innerHTML = ICONS.trash;
      remove.onclick = event => {
        event.stopPropagation();
        deleteConversation(conversation.id);
      };

      item.append(open, remove);
      railList.appendChild(item);
    }
  }

  function beginRename(conversation, anchor) {
    const field = document.createElement("input");
    field.type = "text";
    field.className = "chatrename";
    field.value = conversation.title;
    field.setAttribute("aria-label", "Chat name");
    anchor.replaceWith(field);
    field.focus();
    field.select();

    let settled = false;
    const finish = save => {
      if (settled) return;
      settled = true;
      const next = field.value.trim();
      if (save && next && next !== conversation.title) {
        conversation.title = next;
        conversation.titleLocked = true;
        persist();
        syncChrome();
      }
      renderRail();
    };

    field.onkeydown = event => {
      event.stopPropagation();
      if (event.key === "Enter") {
        event.preventDefault();
        finish(true);
      } else if (event.key === "Escape") {
        event.preventDefault();
        finish(false);
      }
    };
    field.onblur = () => finish(true);
  }

  function renameActiveConversation() {
    const conversation = activeConversation();
    if (!conversation) return;
    if (rootEl.dataset.rail !== "on") toggleRail(true);
    requestAnimationFrame(() => {
      const row = railList.querySelector(".chatitem.active .chatname");
      if (row) beginRename(conversation, row);
    });
  }

  }

  async function copyText(text) {
    if (navigator.clipboard?.writeText) {
      await navigator.clipboard.writeText(text);
      return;
    }
    const helper = document.createElement("textarea");

  function highlightCode(code, language) {
    const grammar = HL_GRAMMARS[HL_ALIASES[String(language || "").toLowerCase()]];
    if (!grammar || code.length > 60000) return escapeHtml(code);

    let html = "";
    let plain = "";
    let cursor = 0;

    const flush = () => {
      if (!plain) return;
      html += escapeHtml(plain);
      plain = "";
    };

    while (cursor < code.length) {
      let advanced = false;
      for (const rule of grammar) {
        rule.r.lastIndex = cursor;
        const found = rule.r.exec(code);
        if (!found || !found[0]) continue;
        if (rule.t) {
          flush();
          html += `<span class="${rule.t}">${escapeHtml(found[0])}</span>`;
        } else {
          plain += found[0];
        }
        cursor += found[0].length;
        advanced = true;
        break;
      }
      if (!advanced) {
        plain += code[cursor];
        cursor += 1;
      }
    }

    flush();
    return html;
  }

  function enhanceCode(container) {
    for (const pre of container.querySelectorAll("pre")) {

      const source = code?.textContent || pre.textContent || "";

  function peekOf(text) {
    const lines = String(text).split("\n").filter(line => line.trim());
    const last = lines[lines.length - 1] || "";
    return last.length > 120 ? `…${last.slice(-120)}` : last;
  }

  function formatError(error) {
    if (error instanceof HttpError) return error.message;
    if (error instanceof GenerationStoppedError) return `${error.reason}${error.message ? `\n\n${error.message}` : ""}`;
    if (error instanceof StreamInterruptedError) return error.message;
    if (error instanceof TypeError) return `${error.message}\n\nThis page may be blocking the proxy through Content Security Policy, the network may be offline, or the request was interrupted.`;
    return `${error?.name || "Error"}: ${error?.message || String(error)}`;
  }

  class SmoothWriter {
    constructor(container, onPaint, speed = 1) {
      this.container = container;
      this.textNode = document.createTextNode("");
      this.container.appendChild(this.textNode);
      this.onPaint = onPaint;
      this.speed = speed;
      this.pending = "";
      this.paused = false;
      this.running = false;
      this.frameId = 0;
      this.waiters = [];
      this.frame = this.frame.bind(this);
    }

    push(text) {
      if (!text) return;
      this.pending += text;
      this.start();
    }

    start() {
      if (this.running || this.paused || !this.pending) return;
      this.running = true;
      this.frameId = requestAnimationFrame(this.frame);
    }

    frame() {
      this.frameId = 0;
      if (this.paused) {
        this.running = false;
        return;
      }
      const backlog = this.pending.length;
      let count = backlog > 2400 ? 150 : backlog > 1000 ? 80 : backlog > 420 ? 36 : backlog > 160 ? 16 : backlog > 50 ? 7 : backlog > 12 ? 4 : 2;
      count = Math.max(1, Math.round(count * this.speed));
      if (count < backlog && /[\uD800-\uDBFF]/.test(this.pending.charAt(count - 1))) count += 1;
      const next = this.pending.slice(0, count);
      this.pending = this.pending.slice(count);
      this.textNode.appendData(next);
      this.onPaint();
      if (this.pending) this.frameId = requestAnimationFrame(this.frame);
      else {
        this.running = false;
        this.resolveWaiters();
      }
    }

    pause() {
      this.paused = true;
      if (this.frameId) {
        cancelAnimationFrame(this.frameId);
        this.frameId = 0;
        this.running = false;
      }
    }

    resume() {
      this.paused = false;
      this.start();
    }

    clear() {
      if (this.frameId) cancelAnimationFrame(this.frameId);
      this.frameId = 0;
      this.pending = "";
      this.textNode.data = "";
      this.running = false;
      this.resolveWaiters();
    }

    flush() {
      if (this.frameId) cancelAnimationFrame(this.frameId);
      this.frameId = 0;
      if (this.pending) {
        this.textNode.appendData(this.pending);
        this.pending = "";
        this.onPaint();
      }
      this.running = false;
      this.resolveWaiters();
    }

    whenDrained() {
      if (!this.pending && !this.running) return Promise.resolve();
      return new Promise(resolve => this.waiters.push(resolve));
    }

    resolveWaiters() {
      const waiters = this.waiters.splice(0);
      for (const resolve of waiters) resolve();
    }
  }

  function nearBottom() {
    return thread.scrollHeight - thread.scrollTop - thread.clientHeight < 160;

  function pruneRenderedTurns() {

      notice.className = "notice";
      notice.textContent = "Older messages are hidden from view to keep this window responsive. They remain in context.";
      messages.prepend(notice);
    }

  function createAssistantTurn(model, effortOption, onRegenerate) {
    const expectsThoughts = effortOption?.returnsThoughts === true;
    const refs = assistantShell(model?.label || "Assistant", effortOption?.label || "");
    messages.appendChild(refs.node);
    pruneRenderedTurns();
    syncEmpty();
    scheduleScroll(true);

    const startedAt = performance.now();
    let phase = 0;
    let sawThought = false;
    let transitioned = false;
    let phaseStartedAt = performance.now();
    let finished = false;

    const timer = setInterval(() => {
      refs.elapsed.textContent = `${((performance.now() - startedAt) / 1000).toFixed(1)}s`;
    }, 100);

    const thoughtWriter = new SmoothWriter(refs.thinkBody, () => {
      refs.thinkBody.scrollTop = refs.thinkBody.scrollHeight;
      refs.thinkPeek.textContent = peekOf(refs.thinkBody.textContent);
      if (refs.think.classList.contains("open")) scheduleScroll();
    }, 1.5);

    const answerWriter = new SmoothWriter(refs.answer, () => scheduleScroll(), 1);
    answerWriter.pause();

    async function transitionToAnswer(currentPhase) {
      await thoughtWriter.whenDrained();
      if (currentPhase !== phase || finished) return;
      await delay(sawThought ? 150 : expectsThoughts ? 300 : 60);
      if (currentPhase !== phase || finished) return;
      refs.think.classList.remove("live");
      if (sawThought) {
        refs.think.hidden = false;
        refs.thinkLabel.textContent = "Reasoning";
        refs.thinkMeta.textContent = `${((performance.now() - phaseStartedAt) / 1000).toFixed(1)}s`;
      } else {
        refs.think.hidden = true;
      }
      refs.answer.classList.add("live");
      refs.runState.textContent = "Writing";
      answerWriter.resume();
      scheduleScroll(true);
    }

    function startPhase(label) {
      phase += 1;
      sawThought = false;
      transitioned = false;
      phaseStartedAt = performance.now();
      answerWriter.pause();
      refs.wait.hidden = false;
      refs.think.hidden = !expectsThoughts;
      refs.think.classList.toggle("live", expectsThoughts);
      refs.thinkLabel.textContent = phase === 1 ? "Reasoning" : "Reasoning again";
      refs.thinkMeta.textContent = expectsThoughts ? "working" : "";
      refs.runState.textContent = label || (expectsThoughts ? "Reasoning" : "Preparing");
      if (phase > 1) thoughtWriter.push(`\n\n— ${label || "Continuing"} —\n\n`);
    }

    function settle() {
      finished = true;
      refs.wait.hidden = true;
      clearInterval(timer);
      refs.think.classList.remove("live");
      refs.answer.classList.remove("live");
      refs.node.classList.add("settled");
      if (!refs.thinkBody.textContent.trim()) refs.think.hidden = true;
      else refs.thinkMeta.textContent = `${((performance.now() - phaseStartedAt) / 1000).toFixed(1)}s`;
    }

    startPhase(expectsThoughts ? "Reasoning" : "Preparing");

    return {
      node: refs.node,
      refs,
      thoughtText: () => refs.thinkBody.textContent || "",
      pushThought(text) {
        refs.wait.hidden = true;
        sawThought = true;
        refs.think.hidden = false;
        refs.think.classList.add("live");
        refs.runState.textContent = "Reasoning";
        refs.thinkMeta.textContent = "streaming";
        thoughtWriter.push(text);
      },
      pushAnswer(text) {
        refs.wait.hidden = true;
        answerWriter.push(text);
        if (!transitioned) {
          transitioned = true;
          transitionToAnswer(phase);
        }
      },
      setStatus(text) {
        refs.runState.textContent = text;
      },
      async prepareContinuation(label) {
        await answerWriter.whenDrained();
        refs.answer.classList.remove("live");
        startPhase(label);
      },
      resetForRetry(label) {
        thoughtWriter.clear();
        answerWriter.clear();
        refs.wait.hidden = false;
        refs.answer.classList.remove("live", "rich");
        refs.error.classList.remove("on");
        refs.actions.replaceChildren();
        refs.node.classList.remove("settled");
        phase = 0;
        startPhase(label);
      },
      async complete(answer, note, onRegenerateAction) {
        if (!transitioned) {
          transitioned = true;
          transitionToAnswer(phase);
        }
        await thoughtWriter.whenDrained();
        await answerWriter.whenDrained();
        settle();
        refs.elapsed.textContent = `${((performance.now() - startedAt) / 1000).toFixed(1)}s`;
        refs.runState.textContent = "";
        setRichText(refs.answer, answer);
        attachAnswerActions(refs, () => answer, onRegenerateAction).textContent = note || "";
        scheduleScroll();
      },
      partial(answer, note, errorText, onRegenerateAction) {
        thoughtWriter.flush();
        answerWriter.resume();
        answerWriter.flush();
        settle();
        refs.runState.textContent = "Incomplete";
        setRichText(refs.answer, answer);
        attachAnswerActions(refs, () => answer, onRegenerateAction).textContent = note || "Partial answer";
        if (errorText) {
          refs.error.textContent = errorText;
          refs.error.classList.add("on");
        }
        scheduleScroll();
      },
      fail(errorText, onRetry) {
        refs.node.classList.add("failed");
        thoughtWriter.flush();
        answerWriter.resume();
        answerWriter.flush();
        settle();
        refs.runState.textContent = "Failed";
        refs.error.textContent = errorText;
        refs.error.classList.add("on");
        refs.actions.replaceChildren();
        if (onRetry) refs.actions.appendChild(miniButton("Try again", ICONS.redo, onRetry));
        scheduleScroll();
      },
      stop(answer, onRetry) {
        if (!answer) refs.node.classList.add("failed");
        thoughtWriter.flush();
        answerWriter.resume();
        answerWriter.flush();
        settle();
        refs.runState.textContent = "Stopped";
        refs.actions.replaceChildren();
        if (answer) {
          setRichText(refs.answer, answer);
          attachAnswerActions(refs, () => answer, onRegenerate).textContent = "Stopped by you";
        } else {
          refs.error.textContent = "Stopped before an answer was returned.";
          refs.error.classList.add("on");
          if (onRetry) refs.actions.appendChild(miniButton("Try again", ICONS.redo, onRetry));
        }
        scheduleScroll();
      }
    };
  }

  /* ---------- run loop ---------- */

  function submit(text, parentId) {
    const clean = String(text).trim();
    if (!clean || busy) return;
    if (!selectedModelConfig() || !selectedEffortConfig()) {
      toast("No model is available right now");
      return;
    }

    let conversation = activeConversation();
    if (!conversation) conversation = newConversation(true);

    const anchor = parentId === undefined ? conversation.activeLeafId : parentId;
    const userNode = addNode(conversation, anchor || null, { role: "user", text: clean });
    if (!userNode.parentId && conversation.rootIds.length === 1 && !conversation.titleLocked) conversation.title = titleFrom(clean);
    touchConversation(conversation);
    runAssistant(conversation, userNode);
  }

  async function runAssistant(conversation, userNode) {
    const model = selectedModelConfig();
    const effort = selectedEffortConfig();
    if (!model || !effort) {
      toast("No model is available right now");
      return;
    }

    conversation.activeLeafId = userNode.id;
    renderThread();
    scheduleScroll(true);

    const priorNodes = userNode.parentId ? nodePath(conversation, userNode.parentId) : [];
    const trimmed = trimHistory(priorNodes.map(node => ({
      role: node.role,
      text: node.text,
      ...(node.state ? { state: node.state } : {})
    })));
    if (trimmed.removed) toast("Oldest context was trimmed to fit");

    const parentId = userNode.parentId;
    const text = userNode.text;
    const retryHandler = () => submit(text, parentId);

    const assistantView = createAssistantTurn(model, effort, () => {
      const leaf = conversation.nodes[conversation.activeLeafId];
      if (leaf?.role === "assistant") regenerate(leaf.id);
    });

    let fullAnswer = "";
    let recoveryCount = 0;
    controller = new AbortController();
    setBusy(true);

    try {
      let result;
      while (true) {
        try {
          assistantView.setStatus(recoveryCount ? `Retrying ${recoveryCount}/${SETTINGS.maxNetworkRetries}` : "Working");
          result = await openProxyRequest(trimmed.items, text, model.id, effort.id, controller.signal);
          break;
        } catch (error) {
          if (error?.name === "AbortError") throw error;
          if (!isRetryableError(error) || recoveryCount >= SETTINGS.maxNetworkRetries) throw error;
          recoveryCount += 1;
          const waitMilliseconds = retryDelayMs(error, recoveryCount);
          const waitSeconds = Math.max(1, Math.ceil(waitMilliseconds / 1000));
          assistantView.resetForRetry(
            error instanceof HttpError && error.status === 429
              ? `Rate limited · retrying in ${waitSeconds}s`
              : `Retrying ${recoveryCount}/${SETTINGS.maxNetworkRetries}`
          );
          await delay(waitMilliseconds, controller.signal);
        }
      }

      if (result.model?.label) {
        modelLine.textContent = result.model.provider ? `${result.model.label} · ${result.model.provider}` : result.model.label;
      }

      if (result.thought) assistantView.pushThought(result.thought);
      if (result.answer) {
        fullAnswer = result.answer;
        assistantView.pushAnswer(result.answer);
      }

      if (!fullAnswer.trim()) {
        throw new GenerationStoppedError(
          result.finishReason || "NO_FINAL_ANSWER",
          result.finishMessage || "The server finished without returning an answer."
        );
      }

      const note = [
        result.outputTokens ? `${result.outputTokens.toLocaleString()} output tokens` : "",
        result.reasoningTokens ? `${result.reasoningTokens.toLocaleString()} reasoning tokens` : "",
        result.continuations ? `${result.continuations} continuation${result.continuations === 1 ? "" : "s"}` : "",
        result.finishReason && result.finishReason !== "STOP" ? result.finishReason : ""
      ].filter(Boolean).join(" · ");

      const assistantNode = addNode(conversation, userNode.id, {
        role: "assistant",
        text: fullAnswer,
        thought: (result.thought || "").slice(0, MAX_STORED_THOUGHT),
        note,
        modelLabel: model.label,
        effortLabel: effort.label,
        state: result.state || null
      });
      assistantView.node.dataset.turn = assistantNode.id;
      touchConversation(conversation);
      syncChrome();

      if (result.finishReason && result.finishReason !== "STOP") {
        assistantView.partial(
          fullAnswer,
          note || "Partial answer",
          result.finishMessage || `Generation finished with ${result.finishReason}.`,
          () => regenerate(assistantNode.id)
        );
      } else {
        await assistantView.complete(fullAnswer, note, () => regenerate(assistantNode.id));
      }

      const switcher = branchSwitcher(conversation, assistantNode);
      if (switcher) assistantView.refs.actions.prepend(switcher);
    } catch (error) {
      markUserStale(userNode.id, "not in context");
      dropNode(conversation, userNode.id);
      persist();
      if (error?.name === "AbortError") assistantView.stop(fullAnswer, retryHandler);
      else assistantView.fail(formatError(error), retryHandler);
    } finally {
      controller = null;
      setBusy(false);
      if (!compact) input.focus();
    }
  }

  }

  function clearFind() {
    for (const mark of findMarks) {
      const parent = mark.parentNode;

      parent.normalize();

  function focusMatch(index) {
    if (!findMarks.length) return;
    findAt = (index + findMarks.length) % findMarks.length;
    findMarks.forEach((mark, position) => mark.classList.toggle("on", position === findAt));
    findMarks[findAt].scrollIntoView({ block: "center" });
    findCount.textContent = `${findAt + 1} of ${findMarks.length}`;
  }

  function runFind(keepPosition) {
    const previous = findAt;
    clearFind();
    const needle = findInput.value.trim().toLowerCase();
    if (!needle) {
      findCount.textContent = "";
      return;
    }

    const walker = document.createTreeWalker(messages, NodeFilter.SHOW_TEXT, {
      acceptNode(node) {
        if (!node.nodeValue.trim()) return NodeFilter.FILTER_REJECT;
        if (node.parentElement?.closest(".uactions,.aactions,.thinkhead,.codebar,.branch,.notice")) return NodeFilter.FILTER_REJECT;
        return NodeFilter.FILTER_ACCEPT;
      }
    });

    const targets = [];
    let node = walker.nextNode();
    while (node) {
      targets.push(node);
      node = walker.nextNode();
    }

    for (const textNode of targets) {
      let current = textNode;
      while (current) {
        const found = current.nodeValue.toLowerCase().indexOf(needle);
        if (found < 0) break;
        const matched = current.splitText(found);
        const rest = matched.splitText(needle.length);
        const mark = document.createElement("mark");
        mark.className = "hit";
        mark.textContent = matched.nodeValue;
        matched.replaceWith(mark);
        findMarks.push(mark);
        current = rest;
      }
    }

    if (!findMarks.length) {
      findCount.textContent = "no matches";
      return;
    }
    focusMatch(keepPosition ? Math.min(previous, findMarks.length - 1) : 0);
  }

  function openFind() {
    findOpen = true;
    rootEl.classList.add("finding");
    findbar.hidden = false;
    findInput.select();
    findInput.focus();
    if (findInput.value.trim()) runFind(false);
  }

  function closeFind() {
    if (!findOpen) return;
    findOpen = false;
    rootEl.classList.remove("finding");
    clearFind();
    findbar.hidden = true;
    findCount.textContent = "";
    if (!compact) input.focus();
  }

  findInput.addEventListener("input", () => runFind(false));
  findInput.addEventListener("keydown", event => {
    event.stopPropagation();
    if (event.key === "Enter") {
      event.preventDefault();
      focusMatch(findAt + (event.shiftKey ? -1 : 1));
    } else if (event.key === "Escape") {
      event.preventDefault();
      closeFind();
    }
  });

  $("#findNext").onclick = () => focusMatch(findAt + 1);
  $("#findClose").onclick = () => closeFind();

  function fuzzyScore(query, text) {
    if (!query) return 1;
    const needle = query.toLowerCase();
    const hay = text.toLowerCase();
    const direct = hay.indexOf(needle);
    if (direct === 0) return 1000;
    if (direct > 0) return 800 - Math.min(direct, 200);
    let cursor = 0;
    let score = 0;
    let previous = -2;
    for (let index = 0; index < hay.length && cursor < needle.length; index += 1) {
      if (hay[index] !== needle[cursor]) continue;
      score += previous === index - 1 ? 6 : 2;
      if (index === 0 || /[\s\-_/·]/.test(hay[index - 1])) score += 4;
      previous = index;
      cursor += 1;
    }
    return cursor === needle.length ? score : 0;
  }

  function agoLabel(timestamp) {
    const seconds = Math.max(1, Math.round((Date.now() - timestamp) / 1000));
    if (seconds < 90) return "just now";
    const minutes = Math.round(seconds / 60);
    if (minutes < 60) return `${minutes}m ago`;
    const hours = Math.round(minutes / 60);
    if (hours < 24) return `${hours}h ago`;
    return `${Math.round(hours / 24)}d ago`;
  }

  function renderPalette() {
    const query = palInput.value.trim();
    palItems = paletteCommands()
      .map(command => ({ command, score: fuzzyScore(query, `${command.label} ${command.hint || ""}`) }))
      .filter(entry => entry.score > 0)
      .sort((a, b) => b.score - a.score)
      .map(entry => entry.command);

    if (palIndex >= palItems.length) palIndex = Math.max(0, palItems.length - 1);
    palList.replaceChildren();
    palCount.textContent = palItems.length ? `${palItems.length} result${palItems.length === 1 ? "" : "s"}` : "";

    if (!palItems.length) {
      const blank = document.createElement("div");
      blank.className = "palempty";
      blank.textContent = "Nothing matches that.";
      palList.appendChild(blank);
      return;
    }

    let lastGroup = "";
    palItems.forEach((command, index) => {
      if (command.group !== lastGroup) {
        lastGroup = command.group;
        const heading = document.createElement("div");
        heading.className = "palgroup";
        heading.textContent = command.group;
        palList.appendChild(heading);
      }
      const item = document.createElement("button");
      item.type = "button";
      item.className = `palitem${index === palIndex ? " on" : ""}`;
      item.setAttribute("role", "option");
      item.innerHTML = `<span class="pi" aria-hidden="true">${command.icon}</span><span class="pallabel"></span>${command.hint ? '<span class="palhint"></span>' : ""}`;
      item.querySelector(".pallabel").textContent = command.label;
      if (command.hint) item.querySelector(".palhint").textContent = command.hint;
      item.onmousemove = () => {
        if (palIndex === index) return;
        palIndex = index;
        highlightPalette();
      };
      item.onclick = () => runPalette(index);
      palList.appendChild(item);
    });
  }

  function movePalette(delta) {
    if (!palItems.length) return;
    palIndex = (palIndex + delta + palItems.length) % palItems.length;
    highlightPalette();
  }

  function runPalette(index) {
    const command = palItems[index === undefined ? palIndex : index];
    if (!command) return;
    closePalette();
    command.run();
  }

  function openPalette() {
    if (paletteOpen) {
      closePalette();
      return;
    }
    closeMenu();
    paletteOpen = true;
    paletteEl.hidden = false;
    palInput.value = "";
    palIndex = 0;
    renderPalette();
    palInput.focus();
  }

  function closePalette() {
    if (!paletteOpen) return;
    paletteOpen = false;
    paletteEl.hidden = true;
    palList.replaceChildren();
    palItems = [];
    if (!compact) input.focus();
  }

  palInput.addEventListener("input", () => {
    palIndex = 0;
    renderPalette();
  });

  palInput.addEventListener("keydown", event => {
    if (event.key === "ArrowDown") {
      event.preventDefault();
      movePalette(1);
    } else if (event.key === "ArrowUp") {
      event.preventDefault();
      movePalette(-1);
    } else if (event.key === "Enter" && !event.isComposing) {
      event.preventDefault();
      runPalette();
    } else if (event.key === "Escape") {
      event.preventDefault();
      event.stopPropagation();
      closePalette();
    }
  });

  paletteEl.addEventListener("pointerdown", event => {
    if (event.target === paletteEl) closePalette();
  });

  /* ---------- controls ---------- */

  const SUGGESTIONS = [
    "Explain this in plain language",
    "Rewrite this to be clearer",
    "Find the bug in this code",
    "Turn these notes into a plan"
  ];

  for (const suggestion of SUGGESTIONS) {
    const chip = document.createElement("button");
    chip.type = "button";
    chip.className = "chip";
    chip.textContent = suggestion;
    chip.onclick = () => {
      input.value = `${suggestion}: `;
      resizeInput();
      syncCount();
      updateSendState();
      input.focus();
      input.setSelectionRange(input.value.length, input.value.length);
    };
    chipsEl.appendChild(chip);
  }

  $("#btnClose").onclick = () => closeApp();
  $("#btnCloseMobile").onclick = () => closeApp();
  $("#btnMin").onclick = () => setMinimized(true);
  $("#btnZoom").onclick = () => toggleZoom();
  dock.onclick = () => setMinimized(false);
  railToggle.onclick = () => toggleRail();
  railDim.onclick = () => toggleRail(false);
  sendButton.onclick = sendCurrentInput;

  $("#btnNew").onclick = () => {
    if (busy) {
      toast("Wait for the current answer to finish");
      return;
    }
    const current = activeConversation();
    if (current && !Object.keys(current.nodes).length) {
      toast("This chat is already empty");
      if (compact) toggleRail(false);
      input.focus();
      return;
    }
    if (current) current.draft = input.value;
    newConversation(true);
    if (compact) toggleRail(false);
    input.focus();
  };

  modelBtn.onclick = () => {
    if (!serverConfig) return;
    openMenu(modelBtn, "Model", serverConfig.models.map(model => ({
      id: model.id,
      label: model.label,
      description: model.provider,
      selected: model.id === modelId
    })), item => applyModel(item.id, true));
  };

  effortBtn.onclick = () => {
    const model = selectedModelConfig();
    if (!model) return;
    openMenu(effortBtn, "Thinking effort", model.thinking.map(option => ({
      id: option.id,
      label: option.label,
      description: option.description,
      selected: option.id === effortId
    })), item => applyEffort(item.id, true));
  };

  input.addEventListener("input", () => {
    resizeInput();
    syncCount();
    updateSendState();
    const conversation = activeConversation();
    if (conversation) {
      conversation.draft = input.value;
      persist();
    }
  });

  input.addEventListener("keydown", event => {
    if (event.key === "Enter" && !event.shiftKey && !event.isComposing && configReady) {
      event.preventDefault();
      sendCurrentInput();
    }
  });

  root.addEventListener("keydown", event => {
    const meta = event.metaKey || event.ctrlKey;
    if (event.key === "Escape") {
      event.preventDefault();
      if (paletteOpen) {
        closePalette();
        return;
      }
      if (findOpen) {
        closeFind();
        return;
      }
      if (openMenuButton) {
        closeMenu();
        return;
      }
      if (compact && rootEl.dataset.rail === "on") {
        toggleRail(false);
        return;
      }
      if (!compact && mode === "max") {
        toggleZoom();
        return;
      }
      closeApp();
      return;
    }
    if (meta && event.key.toLowerCase() === "b") {
      event.preventDefault();
      toggleRail();
    }
    if (meta && event.key.toLowerCase() === "k") {
      event.preventDefault();
      openPalette();
    }
    if (meta && event.key.toLowerCase() === "f") {
      event.preventDefault();
      openFind();
    }
    if (meta && event.key === "Enter") {
      event.preventDefault();
      sendCurrentInput();
    }
  });

  for (const eventName of ["keydown", "keyup", "keypress", "pointerdown", "pointerup", "click", "wheel", "touchstart", "touchmove"]) {
    host.addEventListener(eventName, event => event.stopPropagation());
  }

  /* ---------- lifecycle ---------- */

  const onViewportChange = () => syncViewport();
  addEventListener("resize", onViewportChange, { passive: true });

  function closeApp() {
    if (closing) return;
    closing = true;
    closePalette();
    closeFind();
    const conversation = activeConversation();
    if (conversation) conversation.draft = input.value;
    clearTimeout(persistTimer);
    writeConversations();
    saveWindow();
    controller?.abort();
    cancelAnimationFrame(scrollFrame);
    removeEventListener("resize", onViewportChange);
    winEl.classList.add("leave");
    setTimeout(() => host.remove(), 200);
  }

  host.__gemmaClose = closeApp;

  conversations = sanitizeConversations(store.get("conversations", []));
  activeId = store.get("activeId", "");
  if (!conversations.some(item => item.id === activeId)) activeId = conversations[0]?.id || "";
  if (!activeId) newConversation(true);

  if (!store.usable) {
    emptyEyebrow.textContent = "Temporary session";
    emptyNote.textContent = "This site blocks local storage, so chats in this window will not be saved.";
    footText.textContent = "Not saved on this site";
  }

  syncViewport();
  paint(false);
  renderRail();
  renderThread({ restoreDraft: true });
  setStatus("Loading models");
  setBusy(false);
  setTimeout(() => winEl.classList.remove("enter"), 470);
  loadServerConfig();
})();
